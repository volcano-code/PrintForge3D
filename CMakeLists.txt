cmake_minimum_required(VERSION 3.20)
project(printforge3d LANGUAGES CXX)
option(PRINTFORGE_BUILD_PYTHON "Build optional pybind11 extension" OFF)
option(PRINTFORGE_ENABLE_OPENMP "Enable OpenMP where available" ON)
add_library(printforge_core cpp/core/geometry.cpp)
target_include_directories(printforge_core PUBLIC cpp/include)
target_compile_features(printforge_core PUBLIC cxx_std_17)
set_target_properties(printforge_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(printforge_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(printforge_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(PRINTFORGE_ENABLE_OPENMP)
  find_package(OpenMP QUIET)
  if(OpenMP_CXX_FOUND)
    target_link_libraries(printforge_core PUBLIC OpenMP::OpenMP_CXX)
  endif()
endif()
if(PRINTFORGE_BUILD_PYTHON)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_printforge cpp/bindings/module.cpp)
  target_link_libraries(_printforge PRIVATE printforge_core)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(printforge_core_test cpp/tests/test_geometry.cpp)
  target_link_libraries(printforge_core_test PRIVATE printforge_core)
  add_test(NAME printforge_core_test COMMAND printforge_core_test)
endif()
